import discord from discord.ext import commands import os from dotenv import load_dotenv # 載入 .env 變數 load_dotenv() TOKEN = os.getenv('DISCORD_TOKEN') class MyBot(commands.Bot): def __init__(self): intents = discord.Intents.default() intents.message_content = True super().__init__(command_prefix='!', intents=intents) # 啟動時自動載入 cogs 資料夾內的所有模組 async def setup_hook(self): # 取得 main.py 所在的絕對目錄路徑 (這行是修復報錯的關鍵！) base_dir = os.path.dirname(os.path.abspath(__file__)) cogs_dir = os.path.join(base_dir, 'cogs') # 讀取絕對路徑下的 cogs 資料夾 for filename in os.listdir(cogs_dir): if filename.endswith('.py'): await self.load_extension(f'cogs.{filename[:-3]}') print(f"📦 已載入模組: {filename}") # ⚠️ 修正縮排：將 on_ready 放進 MyBot 類別內 async def on_ready(self): print(f'✅ 機器人 {self.user} 已經成功連線並上線！') # 發送上線公告 embed = discord.Embed( title="🤖 系統已上線！包含多媒體與量化金融模組", description=( "以下是目前可以運行的所有指令列表：\n\n" "**一般指令**\n" "🧹 `!clear`\n\n" "**多媒體指令**\n" "🔮 `!ytsub` | 🎧 `!yt_whisper` | 🎵 `!ytmp3` | 🎬 `!ytmp4`\n\n" "**雲端整合**\n" "☁️ `!dbupload` - 將本地 MP3 上傳至 Dropbox 並產出 HTML 播放列表\n\n" "**量化金融**\n" "📥 `!futudl` | 🚨 `!futuscan` | 🔄 `!sync_stock` | 🔄 `!sync_index` | 🎯 `!warrant`\n\n" "**會計自動化 (應收帳款與銀行)**\n" "📦 `!download_and_scan` | 📊 `!check_and_reconcile` | 🏦 `!update_bank`\n\n" "**費用報銷 (應付帳款)**\n" "🧾 `!process_claims` - 掃描並建檔同事的新報銷單據\n" "🏷️ `!rename_claims` - 自動將實體檔案改名為 AI 建議的標準檔名" ), color=discord.Color.blue() ) embed.set_footer(text="系統模組化升級完成 | Cogs 架構運行中") for guild in self.guilds: for channel in guild.text_channels: # 檢查機器人是否有權限在該頻道發送訊息 if channel.permissions_for(guild.me).send_messages: try: await channel.send(embed=embed) break except Exception as e: print(f"❌ 無法在伺服器 [{guild.name}] 發送訊息: {e}") bot = MyBot() if __name__ == "__main__": bot.run(TOKEN)
